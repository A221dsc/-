cmake_minimum_required(VERSION 3.20)
project(spatial_index_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.75 REQUIRED)
find_path(LSI_INCLUDE spatialindex/capi/sidx_api.h REQUIRED)
find_library(LSI_LIBRARY NAMES spatialindex_c spatialindex_c-64 REQUIRED)
add_library(indexes src/indexes/grid.cpp src/indexes/quad.cpp
                    src/indexes/boost.cpp src/indexes/lsi.cpp)
target_include_directories(indexes PUBLIC src PRIVATE ${LSI_INCLUDE})
target_link_libraries(indexes PRIVATE Boost::headers ${LSI_LIBRARY})
add_library(oracle src/validation/oracle.cpp)
target_include_directories(oracle PUBLIC src)
enable_testing()
add_executable(index_tests tests/index_tests.cpp)
target_link_libraries(index_tests PRIVATE indexes oracle)
add_test(NAME index_correctness COMMAND index_tests)
add_executable(spatial_bench src/benchmark/main.cpp src/workload/replay.cpp)
target_link_libraries(spatial_bench PRIVATE indexes oracle)
if(WIN32)
  target_link_libraries(spatial_bench PRIVATE psapi)
endif()
